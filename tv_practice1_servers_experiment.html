Например, загрузка возникает ровно одним способом – все заявки попали на сервер A. А загрузка при фиксированном порядке серверов возникает способами. Поэтому нельзя просто считать подходящие тройки.

Код эксперимента

Ту же симуляцию можно провести отдельно на Python.
Показать код на Python 
import random

THEORY = 50 / 81


def one_experiment():
    counts = [0, 0, 0]

    for _ in range(5):
        server = random.randrange(3)
        counts[server] += 1

    balanced = all(1 <= x <= 3 for x in counts)
    return counts, balanced


def simulate(n):
    balanced_count = 0

    for _ in range(n):
        _, balanced = one_experiment()
        balanced_count += balanced

    frequency = balanced_count / n
    error = abs(frequency - THEORY)

    return frequency, error, balanced_count


def fmt(x):
    return f"{x:.4f}".replace(".", ",")


for n in (100, 1000, 10_000):
    frequency, error, balanced_count = simulate(n)

    print(
        f"N = {n:>6}: "
        f"частота = {fmt(frequency)}, "
        f"ошибка = {fmt(error)}, "
        f"сбалансировано = {balanced_count}"
    )
